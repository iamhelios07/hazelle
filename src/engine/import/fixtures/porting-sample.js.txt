// Draws a small picture for each reply and puts it on that reply, with 다시 그리기 in every message's menu, in RisuAI
// and in Hazelle alike. What would touch RisuAI's page goes through PluginSlots: Hazelle has it built in, and on RisuAI
// the piece Hazelle publishes, pasted above this, fills it. Without it, the plugin does nothing rather than fail.

async function draw(seed) {
  const canvas = new OffscreenCanvas(96, 96);
  const context = canvas.getContext('2d');
  context.fillStyle = `hsl(${(seed * 37) % 360} 70% 55%)`;
  context.fillRect(0, 0, 96, 96);
  const blob = await canvas.convertToBlob({ type: 'image/png' });
  return new Uint8Array(await blob.arrayBuffer());
}

// Not awaited where it is called: drawing may take as long as it takes, and the chat goes on meanwhile.
function attach(chatId, seed) {
  (async () => PluginSlots.attachPicture(chatId, await draw(seed), { caption: '응답에 맞춘 색' }))().catch((error) => console.error(error));
}

(async () => {
  if (typeof PluginSlots === 'undefined') return;

  await Risuai.addRisuChatListener('output', async ({ characterIndex, chatIndex, messageIndex }) => {
    const chat = await Risuai.getChatFromIndex(characterIndex, chatIndex);
    const reply = chat?.message[messageIndex];
    if (reply) attach(reply.chatId, reply.data.length);
  });

  await PluginSlots.registerMessageAction({ id: 'redraw', label: '다시 그리기' }, ({ chatId }) => attach(chatId, Date.now()));
})();
